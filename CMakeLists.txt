cmake_minimum_required(VERSION 3.24)
project(Tailor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Svg Test)
qt_standard_project_setup(REQUIRES 6.8)
option(TAILOR_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(MSVC)
    add_compile_options(/W4 /permissive-)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(TAILOR_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
qt_add_library(tailor_core STATIC src/Firmware.cpp src/JLink.cpp)
target_include_directories(tailor_core PUBLIC src)
target_link_libraries(tailor_core PUBLIC Qt6::Core)
qt_add_executable(tailor WIN32 MACOSX_BUNDLE src/main.cpp src/MainWindow.cpp
    src/theme/ThemeManager.cpp src/theme/MdiIcons.cpp src/theme/PlatformChrome.cpp)
if(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(resources/icons/tailor.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(tailor PRIVATE src/theme/PlatformChrome_mac.mm resources/icons/tailor.icns)
    target_link_libraries(tailor PRIVATE "-framework AppKit")
    set_target_properties(tailor PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.nex8er.tailor
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
endif()
if(WIN32)
    target_sources(tailor PRIVATE resources/icons/tailor.rc)
    target_include_directories(tailor PRIVATE resources/icons)
endif()
target_include_directories(tailor PRIVATE include src/theme)
target_link_libraries(tailor PRIVATE tailor_core Qt6::Widgets Qt6::Svg)
qt_add_resources(tailor assets PREFIX "/tailor" BASE resources FILES
    resources/fonts/materialdesignicons-webfont.ttf resources/themes/spotty.qss
    resources/icons/chevron-down.svg resources/icons/chevron-up.svg
    resources/icons/chevron-down-muted.svg resources/icons/chevron-up-muted.svg resources/icons/check.svg
    resources/icons/tailor.png)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(fake_jlink tests/FakeJLink.cpp)
    target_link_libraries(fake_jlink PRIVATE Qt6::Core)
    qt_add_executable(tailor_tests tests/CoreTest.cpp)
    add_dependencies(tailor_tests fake_jlink)
    target_compile_definitions(tailor_tests PRIVATE FAKE_JLINK_PATH="$<TARGET_FILE:fake_jlink>")
    target_link_libraries(tailor_tests PRIVATE tailor_core Qt6::Test)
    add_test(NAME core COMMAND tailor_tests)
    qt_add_executable(tailor_ui_tests tests/UiTest.cpp src/MainWindow.cpp
        src/theme/ThemeManager.cpp src/theme/MdiIcons.cpp src/theme/PlatformChrome.cpp)
    target_include_directories(tailor_ui_tests PRIVATE include src/theme)
    target_link_libraries(tailor_ui_tests PRIVATE tailor_core Qt6::Widgets Qt6::Svg Qt6::Test)
    if(APPLE)
        target_sources(tailor_ui_tests PRIVATE src/theme/PlatformChrome_mac.mm)
        target_link_libraries(tailor_ui_tests PRIVATE "-framework AppKit")
    endif()
    qt_add_resources(tailor_ui_tests test_assets PREFIX "/tailor" BASE resources FILES
        resources/fonts/materialdesignicons-webfont.ttf resources/themes/spotty.qss
        resources/icons/chevron-down.svg resources/icons/chevron-up.svg
        resources/icons/chevron-down-muted.svg resources/icons/chevron-up-muted.svg resources/icons/check.svg)
    add_test(NAME ui COMMAND tailor_ui_tests)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

endif()
install(TARGETS tailor BUNDLE DESTINATION . RUNTIME DESTINATION bin)
install(FILES README.md resources/fonts/LICENSE-MDI.txt DESTINATION share/tailor)
if(WIN32)
    install(FILES packaging/register-windows.ps1 DESTINATION .)
endif()
if(UNIX AND NOT APPLE)
    install(FILES packaging/tailor.desktop DESTINATION share/applications)
    install(FILES packaging/tailor.xml DESTINATION share/mime/packages)
    foreach(size IN ITEMS 16 32 48 64 128 256 512)
        install(FILES resources/icons/tailor-${size}.png
            DESTINATION share/icons/hicolor/${size}x${size}/apps RENAME tailor.png)
    endforeach()
endif()
qt_generate_deploy_app_script(TARGET tailor OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
